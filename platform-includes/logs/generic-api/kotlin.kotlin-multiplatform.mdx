```kotlin
import io.sentry.kotlin.multiplatform.Sentry
import io.sentry.kotlin.multiplatform.log.SentryLogLevel

fun logRequestResult(statusCode: Int, endpoint: String) {
    val level = if (statusCode >= 500) SentryLogLevel.ERROR else SentryLogLevel.INFO

    Sentry.logger.log(level, "Request to %s completed", endpoint) {
        this["http.response.status_code"] = statusCode
    }
}
```
