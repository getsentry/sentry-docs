captureCtx, scope := sentry.WithScope(ctx)
scope.SetTag("my-tag", "my value")
scope.SetLevel(sentry.LevelWarning)
// This capture includes the local tag and level.
sentry.CaptureException(captureCtx, errors.New("my error"))

// The parent context does not include the local changes.
sentry.CaptureException(ctx, errors.New("another error"))
