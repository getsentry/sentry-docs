// Add this to your SentryOptionsConfiguration.Configure method
options.SetBeforeSend(@event =>
{
    // SaveFileCorruptedException is an exception defined by your game
    if (@event.Exception is SaveFileCorruptedException)
    {
        @event.SetFingerprint(new[] { "save-file-corrupted" });
    }

    return @event;
});
